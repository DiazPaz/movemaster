cmake_minimum_required(VERSION 3.16)
project(movemaster_hardware VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MOVEMASTER_BUILD_ROS2 "Build ros2_control plugin (ROS 2 Jazzy)" ON)
include(CTest)
find_package(nlohmann_json QUIET)
if(NOT nlohmann_json_FOUND)
  find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
  add_library(nlohmann_json::nlohmann_json INTERFACE IMPORTED)
  set_target_properties(nlohmann_json::nlohmann_json PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${NLOHMANN_JSON_INCLUDE_DIR}")
endif()
find_package(Threads REQUIRED)

add_library(sparkmax_protocol SHARED src/sparkmax_json_protocol.cpp)
add_library(movemaster_driver SHARED src/socketcan.cpp src/movemaster_driver.cpp src/driver_config.cpp)
foreach(target sparkmax_protocol movemaster_driver)
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
target_link_libraries(sparkmax_protocol PUBLIC nlohmann_json::nlohmann_json)
target_link_libraries(movemaster_driver PUBLIC sparkmax_protocol Threads::Threads)

add_executable(protocol_demo examples/protocol_demo.cpp)
target_link_libraries(protocol_demo PRIVATE sparkmax_protocol)
add_executable(driver_monitor examples/driver_monitor.cpp)
target_link_libraries(driver_monitor PRIVATE movemaster_driver)

if(BUILD_TESTING)
  add_executable(protocol_oracle tests/protocol_oracle.cpp)
  target_link_libraries(protocol_oracle PRIVATE sparkmax_protocol)
  add_executable(driver_test tests/driver_test.cpp)
  target_link_libraries(driver_test PRIVATE movemaster_driver)
  add_test(NAME driver_fake_bus COMMAND driver_test ${CMAKE_CURRENT_SOURCE_DIR}/spec/spark-frames-2.1.0)
  add_test(NAME protocol_demo_no_can COMMAND protocol_demo ${CMAKE_CURRENT_SOURCE_DIR}/spec/spark-frames-2.1.0)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME python_cpp_parity COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/differential_test.py --oracle $<TARGET_FILE:protocol_oracle>)
  endif()
endif()

if(MOVEMASTER_BUILD_ROS2)
  find_package(ament_cmake REQUIRED)
  find_package(hardware_interface REQUIRED)
  find_package(pluginlib REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_lifecycle REQUIRED)
  add_library(movemaster_hardware SHARED src/movemaster_hardware.cpp)
  target_link_libraries(movemaster_hardware PUBLIC movemaster_driver)
  ament_target_dependencies(movemaster_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle)
  pluginlib_export_plugin_description_file(hardware_interface movemaster_hardware.xml)
  install(TARGETS movemaster_hardware EXPORT movemasterTargets LIBRARY DESTINATION lib)
  ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle nlohmann_json Threads)
  ament_export_targets(movemasterTargets HAS_LIBRARY_TARGET)
endif()
install(TARGETS sparkmax_protocol movemaster_driver EXPORT movemasterTargets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(TARGETS protocol_demo driver_monitor RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY spec config docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md NOTICE.md DESTINATION share/${PROJECT_NAME})
if(MOVEMASTER_BUILD_ROS2)
  ament_package()
endif()
